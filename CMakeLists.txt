cmake_minimum_required(VERSION 3.25)
file(READ "${CMAKE_SOURCE_DIR}/package.json" BETTER_SEARCH_PACKAGE)
string(JSON BETTER_SEARCH_VERSION GET "${BETTER_SEARCH_PACKAGE}" version)
project(better_search VERSION ${BETTER_SEARCH_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BETTER_SEARCH_BUILD_PROFILE "custom" CACHE STRING "Build profile name")
set(BETTER_SEARCH_PORTABLE_CPU 0)
if(BETTER_SEARCH_BUILD_PROFILE MATCHES "^(windows|linux)-cpu$")
  set(BETTER_SEARCH_PORTABLE_CPU 1)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static local runtime" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/llama.cpp EXCLUDE_FROM_ALL)
if(BETTER_SEARCH_PORTABLE_CPU AND (GGML_NATIVE OR GGML_CUDA OR GGML_AVX OR GGML_AVX2 OR GGML_AVX512 OR GGML_FMA OR GGML_F16C OR GGML_BMI2 OR GGML_SSE42 OR GGML_OPENMP))
  message(FATAL_ERROR "Portable CPU releases must not require native CPU extensions, CUDA or the external OpenMP runtime.")
endif()
set(VENDOR "${CMAKE_SOURCE_DIR}/.runtime/vendor")
if(NOT EXISTS "${VENDOR}/sqlite/sqlite3.c")
  message(FATAL_ERROR "Run npm run setup first to download pinned native dependencies.")
endif()
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_DISABLE_ASM ON CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/zstd/build/cmake" "${CMAKE_BINARY_DIR}/zstd" EXCLUDE_FROM_ALL)
add_library(search_sqlite STATIC "${VENDOR}/sqlite/sqlite3.c" "${VENDOR}/vec/sqlite-vec.c")
target_include_directories(search_sqlite PUBLIC "${VENDOR}/sqlite" "${VENDOR}/vec")
target_compile_definitions(search_sqlite PRIVATE SQLITE_ENABLE_FTS5 SQLITE_ENABLE_JSON1 SQLITE_THREADSAFE=1 SQLITE_CORE SQLITE_VEC_STATIC)
set(BETTER_SEARCH_VECTOR_ARCH "" CACHE STRING "Explicit MSVC SIMD architecture for this machine's vector scan")
if(MSVC AND BETTER_SEARCH_VECTOR_ARCH MATCHES "^(AVX2|AVX512)$")
  # Only the distance kernels: SQLite itself keeps precise floating-point semantics.
  set_source_files_properties("${VENDOR}/vec/sqlite-vec.c" PROPERTIES
    COMPILE_OPTIONS "/O2;/arch:${BETTER_SEARCH_VECTOR_ARCH};/fp:fast")
endif()
add_executable(better-search native/main.cpp native/store.cpp native/source.cpp native/rollout.cpp native/engine.cpp native/search.cpp native/embedding.cpp native/settings.cpp native/platform.cpp)
set_property(TARGET better-search PROPERTY CXX_STANDARD 20)
target_include_directories(better-search PRIVATE "${VENDOR}" native)
target_compile_definitions(better-search PRIVATE SQLITE_CORE SQLITE_VEC_STATIC BETTER_SEARCH_VERSION="${PROJECT_VERSION}" BETTER_SEARCH_BUILD_PROFILE="${BETTER_SEARCH_BUILD_PROFILE}" BETTER_SEARCH_PORTABLE_CPU=${BETTER_SEARCH_PORTABLE_CPU})
target_link_libraries(better-search PRIVATE search_sqlite llama libzstd_static)
if(MSVC)
  target_compile_options(better-search PRIVATE /utf-8 /W4 /O2)
  target_compile_definitions(better-search PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(better-search PRIVATE ws2_32 bcrypt crypt32 ole32 shell32 uuid)
else()
  target_compile_options(better-search PRIVATE -Wall -Wextra -O3)
  target_link_libraries(better-search PRIVATE pthread dl)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
  add_executable(better-search-dialog native/dialog.cpp)
  target_link_libraries(better-search-dialog PRIVATE PkgConfig::GTK3)
  add_dependencies(better-search better-search-dialog)
endif()
